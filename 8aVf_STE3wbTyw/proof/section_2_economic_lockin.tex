\section*{Section 2: Economic Lock-In (C2)}

\subsection*{Assumptions Used}

\begin{assumption}[Documented Productivity Gains (A2)]
  AI adoption delivers statistically significant, sustained productivity gains
  across sectors.  Formally, for an organization at integration depth $d \geq 1$,
  let $\Pi_{\mathrm{AI}}(d)$ be annual productivity and $\Pi_0$ the pre-AI
  baseline.  Empirical data establish that
  $\Pi_{\mathrm{AI}}(d) \geq (1 + \gamma)\,\Pi_0$ for some $\gamma > 0$
  (with $\gamma \approx 0.3$--$0.55$ in documented cases).
\end{assumption}

\begin{assumption}[Super-Linear Switching Costs (A4)]
  Removing AI from a pipeline at integration depth $d$ incurs switching cost
  $S(d) = \Omega(d^2)$ — i.e., there exists $c > 0$ such that
  $S(d) \geq c\,d^2$ for all $d \geq 1$.
\end{assumption}

\subsection*{Statement to Prove (C2)}

\begin{theorem}[Economic Lock-In]
  Under Assumptions A2 and A4, once AI integration depth $d$ exceeds a
  computable threshold $d^* > 0$, the net present value of reverting to the
  pre-AI baseline is strictly negative for any positive discount rate $\rho > 0$.
  Consequently, continued AI adoption is the unique economically rational
  strategy at the societal scale, making AI adoption irreversible.
\end{theorem}
