\section*{Section 4: Structural Unavoidability (C4)}

\subsection*{Assumptions Used}

\begin{assumption}[Current Sectoral Penetration (A3)]\label{A3}
  As of 2024, AI systems are operationally embedded in the following four
  societal systems, each in a distinct sector:
  \begin{enumerate}
    \item \textbf{Energy} ($E$): AI manages load-balancing in national
      power grids.  Annual operating budget $B_E \approx \$10^{10}$.
    \item \textbf{Healthcare} ($H$): FDA-cleared AI diagnostic tools in
      $>$40\% of US hospitals.  Annual IT budget $B_H \approx \$10^{11}$.
    \item \textbf{Fiscal} ($F$): AI drives IRS and HMRC fraud detection.
      Annual IT budget $B_F \approx \$10^{9}$.
    \item \textbf{Education} ($D$): AI adaptive platforms serve
      $>$50M students.  Annual EdTech budget $B_D \approx \$10^{9}$.
  \end{enumerate}
  Each system is operationally critical: failure causes measurable harm.
  AI has been in active use in each for $\geq 3$ years, with integration
  depth $d_X(0) \geq 100$.
\end{assumption}

\begin{assumption}[Super-Linear Switching Costs with Budget Calibration (A4)]\label{A4}
  For AI integration depth $d \geq 0$, the switching-cost function $S$
  satisfies:
  \begin{enumerate}
    \item \emph{Strictly increasing:} $d_2 > d_1 \geq 0 \Rightarrow S(d_2) > S(d_1)$;
      $S(0) = 0$.
    \item \emph{Super-linear lower bound:} $\exists\, c > 0$ s.t.\
      $S(d) \geq c\,d^2$ for all $d \geq 1$.
    \item \emph{Depth grows with active use:} For each period of
      continued active use, $d(t+1) \geq d(t) + 1$.
    \item \emph{Budget calibration:} For each system $X \in \{E,H,F,D\}$,
      $c\,d_X(0)^2 \geq B_X$.  (That is, at current depth $d_X(0) \geq 100$,
      switching costs already meet or exceed one year's operating budget.)
  \end{enumerate}
\end{assumption}

\begin{assumption}[No Viable Short-Term Substitute (A4b)]\label{A4b}
  For each system $X \in \{E,H,F,D\}$, no non-AI system can replicate
  the current AI-performed critical functions at acceptable cost and
  quality within a 5-year horizon.
\end{assumption}

\subsection*{Definitions}

\begin{definition}[Sectorally Distinct]\label{defSD}
  Systems are \emph{sectorally distinct} if each belongs to a different
  societal sector (energy, healthcare, fiscal, education are
  four such sectors), and performing a critical function in one sector
  does not constitute performing a critical function in another.
\end{definition}

\begin{definition}[Financially Prohibitive]\label{defFP}
  Removing AI from system $X$ is \emph{financially prohibitive} if the
  one-time switching cost satisfies $S(d_X) \geq B_X$, i.e., the
  removal cost is at least equal to the system's full annual operating
  budget.
\end{definition}

\subsection*{Statement to Prove (C4)}

\begin{theorem}[Structural Embeddedness]\label{thm:C4}
  Under Assumptions~\ref{A3}, \ref{A4}, and~\ref{A4b}:
  \begin{enumerate}
    \item[(i)] AI is operationally embedded in at least four sectorally
      distinct critical systems (Definition~\ref{defSD}).
    \item[(ii)] Removing AI from any of these systems is currently
      financially prohibitive (Definition~\ref{defFP}):
      $S(d_X(0)) \geq B_X$ for every $X \in \{E,H,F,D\}$.
    \item[(iii)] Continued active use of any system $X$ strictly increases
      $S(d_X(t))$ over time: for all $n \geq 1$,
      $S(d_X(t+n)) > S(d_X(t))$ and
      $S(d_X(t+n)) \geq c\,(d_X(t) + n)^2$.
    \item[(iv)] No technically viable short-term alternative to AI exists
      for these systems (A4b), and the cost barrier to removal only
      grows.  AI's role in these systems is thus doubly entrenched: the
      financial cost of removal is already prohibitive and increasing,
      while no substitute is available within 5 years.
  \end{enumerate}
\end{theorem}

\begin{proof}

\noindent\textbf{Step 1: Claim (i) — four sectorally distinct critical systems.}

By A3, AI is embedded in systems $E$ (energy), $H$ (healthcare),
$F$ (fiscal), $D$ (education).  These are four distinct societal sectors
(Definition~\ref{defSD}): delivering electricity management is not the
same as delivering medical diagnosis, tax fraud detection, or education.
Each is operationally critical by A3.  Claim (i) is established.

\medskip
\noindent\textbf{Step 2: Claim (ii) — financially prohibitive switching costs.}

Fix any $X \in \{E,H,F,D\}$.  By A4.2: $S(d_X(0)) \geq c\,d_X(0)^2$.
By A4.4: $c\,d_X(0)^2 \geq B_X$.  Therefore $S(d_X(0)) \geq B_X$.
By Definition~\ref{defFP}, removal of AI from $X$ is financially
prohibitive at current integration depth.  Claim (ii) is established
for all four systems.

\medskip
\noindent\textbf{Step 3: Claim (iii) — strictly increasing switching costs.}

Fix system $X$ in continued active use.  By A4.3: $d_X(t+1) \geq d_X(t)+1$.
By induction: $d_X(t+n) \geq d_X(t) + n$ for all $n \geq 0$.
\emph{Base case} $n=0$: holds trivially.
\emph{Inductive step}: if $d_X(t+n) \geq d_X(t)+n$, then
$d_X(t+n+1) \geq d_X(t+n)+1 \geq d_X(t)+n+1$.

Since $d_X(t+n) \geq d_X(t)+n > d_X(t)$ for $n \geq 1$, and $S$ is
strictly increasing (A4.1):
\[
  S(d_X(t+n)) > S(d_X(t)) \qquad \text{for all } n \geq 1.
\]
By A4.2: $S(d_X(t+n)) \geq c\,d_X(t+n)^2 \geq c\,(d_X(t)+n)^2$,
where the last inequality uses $d_X(t+n) \geq d_X(t)+n \geq 0$,
so $d_X(t+n)^2 \geq (d_X(t)+n)^2$ (by monotonicity of squaring on
non-negatives, since $d_X(t+n) \geq d_X(t)+n \geq 0$).
Claim (iii) is established.

\medskip
\noindent\textbf{Step 4: Claim (iv) — double entrenchment.}

By claim (ii), the current switching cost for each system meets or
exceeds one full year's operating budget.  By claim (iii), this cost
barrier strictly increases as long as the system remains active.
By A4b, no non-AI substitute can replicate the critical functions at
acceptable cost and quality within 5 years.

Therefore: (a) removing AI now costs $\geq B_X$, (b) waiting raises
this cost further, and (c) no viable substitute exists within 5 years.
AI's role in these systems is doubly entrenched: removal faces a
growing financial barrier and no substitute path.  Claim (iv) is
established.

\medskip
\noindent\textbf{Conclusion.}
All four claims are proved from A3, A4 (parts 1--4), and A4b only.
AI is sectorally embedded, faces prohibitive and growing removal costs
in all four critical systems, and has no viable short-term substitute.
\end{proof}
