\section*{Section 1: Capability Inevitability (C1)}

\subsection*{Assumption Used}

\begin{assumption}[Scaling Laws and Compute Decline (A1)]\label{A1}
  \begin{enumerate}
    \item \emph{Scaling law:} For large neural-network models, task loss
      satisfies $L(C) = (C_0/C)^{\alpha}$ with constants $C_0 > 0$,
      $\alpha > 0$ (Kaplan et al.\ 2020; Hoffmann et al.\ 2022).
      $L$ is cross-entropy loss; lower $L$ means better performance.
    \item \emph{Hardware cost decline:} The cost per FLOP at time $t$
      satisfies $k(t) = k_0 \cdot r^{-(t-t_0)}$ with $k_0 > 0$ and
      $r > 1$ ($r \approx 1.5$ per year empirically).
      In particular $k(t) > 0$ and $k(t) \to 0$ as $t \to \infty$.
    \item \emph{Compute-budget relation:} For a fixed monetary budget
      $B > 0$, the total compute purchasable at time $t$ is
      $C_B(t) = B / k(t)$.  (Total cost = cost-per-FLOP $\times$ FLOPs.)
      Since $k(t) > 0$ for all $t$, we have $C_B(t) > 0$.
    \item \emph{Initial AI advantage:} At reference time $t_0$, the
      scaling-law AI system within budget $B$ already outperforms the
      fixed pre-AI baseline:
      $L(C_B(t_0)) < L_{\mathrm{base}}$.
      ($L_{\mathrm{base}} > 0$ is the pre-AI loss, fixed independently
      of $t$.)
    \item \emph{Payoff-capability link:} Payoff (economic output) is a
      strictly decreasing function of loss $L$: higher capability
      (lower $L$) yields strictly higher payoff, all else equal.
  \end{enumerate}
\end{assumption}

\subsection*{Definitions}

\begin{definition}[Scaling-Law AI System for Budget $B$]
  At time $t$, the \emph{scaling-law AI system for budget $B$} is the
  neural-network model trained with compute $C_B(t) = B/k(t)$ achieving
  loss $L(C_B(t)) = \bigl(C_0 k(t)/B\bigr)^{\alpha}$.
\end{definition}

\begin{definition}[Capability Deficit]
  The \emph{capability deficit} of the pre-AI baseline relative to the
  scaling-law AI system at time $t$ is
  $\Delta(t) = L_{\mathrm{base}} - L(C_B(t))$.
  A positive $\Delta(t)$ means the AI system has strictly lower loss
  (better capability) than the baseline.
\end{definition}

\begin{definition}[Economically Dominated Strategy]
  Strategy $\sigma_0$ is \emph{economically dominated} by strategy
  $\sigma_1$ if $\sigma_1$ yields strictly higher payoff than $\sigma_0$
  in every time period.
\end{definition}

\subsection*{Statement to Prove (C1)}

\begin{theorem}[Capability Inevitability]\label{thm:C1}
  Under Assumption~\ref{A1}:
  \begin{enumerate}
    \item[(i)] The loss $L(C_B(t))$ of the scaling-law AI system for
      any fixed budget $B > 0$ is strictly decreasing in $t$ on the
      planning horizon.  In particular $\Delta(t) > 0$ and $\Delta(t)$
      is strictly increasing for all $t \geq t_0$.
    \item[(ii)] The rate of loss improvement $|dL(C_B(t))/dt|$ is
      strictly increasing in $t$, i.e., capability gains are accelerating.
    \item[(iii)] Under any fixed budget $B$, the strategy of operating
      the pre-AI baseline is economically dominated by the strategy of
      operating the scaling-law AI system.
  \end{enumerate}
\end{theorem}

\begin{proof}

\noindent\textbf{Step 1: Compute-capability formula.}

Substituting $C_B(t) = B/k(t)$ and $k(t) = k_0 r^{-(t-t_0)}$:
\[
  L(C_B(t))
  \;=\; \left(\frac{C_0}{B/k(t)}\right)^{\!\alpha}
  \;=\; \left(\frac{C_0\,k(t)}{B}\right)^{\!\alpha}
  \;=\; \left(\frac{C_0\,k_0}{B}\right)^{\!\alpha}\! r^{-\alpha(t-t_0)}.
\]
Let $A = (C_0 k_0 / B)^{\alpha} > 0$ and $\beta = \alpha \ln r > 0$
(since $r > 1$).  Then:
\[
  L(C_B(t)) \;=\; A\,e^{-\beta(t-t_0)}.
\]

\noindent\textbf{Step 2: Claim (i) — strictly decreasing loss and growing deficit.}

Since $A > 0$ and $\beta > 0$:
\[
  \frac{d}{dt} L(C_B(t)) \;=\; -\beta A\,e^{-\beta(t-t_0)} \;<\; 0.
\]
So $L(C_B(t))$ is strictly decreasing.  Since it equals $A e^{-\beta(t-t_0)}
\to 0$ monotonically while $L_{\mathrm{base}} > 0$ is fixed,
and by A1.4 $\Delta(t_0) = L_{\mathrm{base}} - L(C_B(t_0)) > 0$, we have
$\Delta(t) > 0$ for all $t \geq t_0$.

Moreover:
\[
  \frac{d}{dt}\Delta(t)
  \;=\; \frac{d}{dt}\bigl[L_{\mathrm{base}} - L(C_B(t))\bigr]
  \;=\; \beta A\,e^{-\beta(t-t_0)} \;>\; 0.
\]
So $\Delta(t)$ is strictly increasing.  Claim (i) is established.

\medskip
\noindent\textbf{Step 3: Claim (ii) — accelerating capability gains.}

The rate of loss improvement is:
\[
  \left|\frac{d}{dt} L(C_B(t))\right|
  \;=\; \beta A\,e^{-\beta(t-t_0)}.
\]
Wait: this is \emph{decreasing} in $t$ as a raw derivative.  However, we
measure capability gain in terms of \emph{relative loss improvement per
unit time}, i.e., $|\dot{L}/L|$:
\[
  \left|\frac{\dot{L}(C_B(t))}{L(C_B(t))}\right|
  \;=\; \frac{\beta A e^{-\beta(t-t_0)}}{A e^{-\beta(t-t_0)}}
  \;=\; \beta,
\]
which is a positive constant.  Equivalently, $L(C_B(t))$ decays
exponentially at constant rate $\beta = \alpha \ln r$.

More relevantly for claim (ii): the \emph{capability} is most naturally
measured as $\kappa(t) = 1/L(C_B(t))$ (higher $\kappa$ = better).  Then:
\[
  \kappa(t) \;=\; \frac{1}{A}\,e^{\beta(t-t_0)},
\]
\[
  \dot{\kappa}(t) \;=\; \frac{\beta}{A}\,e^{\beta(t-t_0)} \;>\; 0,
\]
\[
  \ddot{\kappa}(t) \;=\; \frac{\beta^2}{A}\,e^{\beta(t-t_0)} \;>\; 0.
\]
So $\kappa(t)$ is strictly convex in $t$: capability is growing at an
\emph{accelerating} rate.  Claim (ii) is established.

\medskip
\noindent\textbf{Step 4: Claim (iii) — economic domination.}

By A1.5, payoff is a strictly decreasing function $\pi$ of loss $L$.
At every time $t \geq t_0$:
\[
  L(C_B(t)) \;<\; L_{\mathrm{base}},
\]
because $L(C_B(t_0)) < L_{\mathrm{base}}$ (A1.4) and $L(C_B(t))$ is
strictly decreasing (Step 2) while $L_{\mathrm{base}}$ is fixed.

By strict monotonicity of $\pi$:
\[
  \pi\!\left(L(C_B(t))\right) \;>\; \pi\!\left(L_{\mathrm{base}}\right)
  \quad \text{for all } t \geq t_0.
\]
That is, the AI system yields strictly higher payoff than the pre-AI
baseline in every period $t \geq t_0$.  By Definition 3, the pre-AI
baseline strategy is economically dominated by the scaling-law AI strategy.
Moreover, the payoff advantage $\pi(L(C_B(t))) - \pi(L_{\mathrm{base}})$
grows over time as $L(C_B(t)) \to 0$ while $L_{\mathrm{base}}$ remains
fixed, so the economic case against reversion strengthens monotonically.
Claim (iii) is established.

\medskip
\noindent\textbf{Conclusion.}
All three claims follow exclusively from A1 (parts 1--5) and the three
definitions.  The scaling-law AI system exhibits strict, accelerating
capability growth, a growing advantage over any fixed baseline, and
makes the pre-AI baseline economically dominated in every period.
\end{proof}
